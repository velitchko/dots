# OPTIONAL laptop extras. Installed only with: ./install.sh --with-laptop
# These ship system services. The installer never enables them; after
# installing, enable what you want yourself:
#   sudo systemctl enable --now bluetooth.service
#   sudo systemctl enable --now power-profiles-daemon.service
bluez
bluez-utils
bluez-qt               # org.kde.bluezqt QML module, for Dolphin's Send via Bluetooth
power-profiles-daemon
# Fingerprint reader: unlock the lock screen (and optionally SDDM, see
# scripts/fingerprint.sh). D-Bus activated, nothing to enable. Enroll: fprintd-enroll
fprintd
# Audio firmware: most laptops from ~2018 on (Intel SOF DSPs) show no sound
# devices at all without these. Harmless on machines that don't need them.
sof-firmware
alsa-ucm-conf
alsa-firmware
